\documentclass[tikz,border=5pt]{standalone}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta,positioning,fit,calc}

\begin{document}
\pagecolor{white}
\begin{tikzpicture}[
    font=\sffamily\small,
    >=Latex,
    mainbox/.style={draw=black!65, rounded corners=2pt, fill=white,
                    align=left, inner xsep=7pt, inner ysep=5pt,
                    minimum width=92mm, text width=86mm},
    outside/.style={mainbox, fill=gray!7, align=center},
    loopbox/.style={draw=blue!70!black, rounded corners=3pt,
                    inner xsep=10pt, inner ysep=9pt},
    badge/.style={circle, fill=blue!70!black, text=white,
                  inner sep=1.2pt, minimum size=5mm,
                  font=\sffamily\bfseries\scriptsize},
    arrow/.style={-{Latex[length=2.8mm,width=1.8mm]},
                  line width=0.75pt, draw=blue!70!black},
    returnarrow/.style={-{Latex[length=2.6mm,width=1.7mm]},
                        line width=0.65pt, draw=black!55},
    smalllabel/.style={font=\sffamily\footnotesize, text=black!65, align=center}
]

\node[outside] (eikonal) at (0,0)
    {\textbf{Relaxed Eikonal solve}\\
     fixed wall distance $y$ for the SA closure};

\node[mainbox, below=9mm of eikonal] (eddy)
    {\textbf{Assemble turbulence coefficients}\\
     Use previous SA field $\tilde{\nu}^{k}$ to form
     $\nu_t^{k}$ and $\nu_{\mathrm{eff}}^{k}=\nu+\nu_t^{k}$.};

\node[mainbox, below=7mm of eddy] (ipcs)
    {\textbf{Advance velocity-pressure state with IPCS}\\
     Tentative velocity $\mathbf{u}^{*}$ $\rightarrow$
     pressure correction $p^{k+1}$ $\rightarrow$
     velocity update $\mathbf{u}^{k+1}$.};

\node[mainbox, below=7mm of ipcs] (sa)
    {\textbf{Solve SA transport equation}\\
     Use $\mathbf{u}^{k+1}$ and $y$ to compute the next working variable
     $\tilde{\nu}^{k+1}\in T_h$.};

\node[mainbox, below=7mm of sa] (relax)
    {\textbf{Relax, bound, and check convergence}\\
     Under-relax $\tilde{\nu}^{k+1}$, recompute $\nu_t^{k+1}$,
     and test the Picard field updates.};

\node[badge] (badge1) at ($(eddy.west)+(-5mm,0)$) {1};
\node[badge] (badge2) at ($(ipcs.west)+(-5mm,0)$) {2};
\node[badge] (badge3) at ($(sa.west)+(-5mm,0)$) {3};
\node[badge] (badge4) at ($(relax.west)+(-5mm,0)$) {4};

\node[loopbox, fit=(badge1)(badge2)(badge3)(badge4)(eddy)(ipcs)(sa)(relax)] (picard) {};
\node[font=\sffamily\bfseries\small, text=blue!70!black, fill=white, inner sep=1.5pt,
      anchor=west] at ($(picard.north west)+(3mm,0)$)
      {outer Picard iteration $k\rightarrow k+1$};

\node[outside, below=9mm of relax, minimum width=48mm, text width=44mm] (stop)
    {converged RANS-SA state};

\draw[arrow] (eikonal) -- (picard.north);
\draw[arrow] (eddy) -- (ipcs);
\draw[arrow] (ipcs) -- (sa);
\draw[arrow] (sa) -- (relax);
\draw[arrow] (relax) -- node[smalllabel, right] {converged} (stop);

\draw[returnarrow] (relax.west) -- ++(-10mm,0)
    |- node[smalllabel, pos=0.25, left] {not\\converged} (eddy.west);

\node[smalllabel, anchor=west] at ($(stop.south west)+(-12mm,-6mm)$)
    {$V_h$: velocity space \quad $Q_h$: pressure space \quad $T_h$: SA working-variable space};

\end{tikzpicture}
\end{document}
