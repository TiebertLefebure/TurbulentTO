% Streamwise lengths are compressed for readability.
\def\xin{0.0}
\def\xwall{1.85}

% Boundary-condition segments.
\draw[inlet] (\xin,\y0) -- (\xin,\yt);
\draw[sym] (\xin,\yt) -- (\xwall,\yt);
\draw[wall] (\xwall,\yt) -- (\xout,\yt);

% Boundary labels.
