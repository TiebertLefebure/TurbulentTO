\documentclass[tikz,border=3mm]{standalone}
\usetikzlibrary{arrows.meta,calc,positioning}
\begin{document}
\pagecolor{white}
\begin{tikzpicture}[
  font=\sffamily\small,
  flow/.style={-{Latex[length=2.8mm,width=1.8mm]}, line width=0.78pt, draw=blue!65!black},
  dim/.style={<->, line width=0.45pt, draw=black!75, shorten <=1.5pt, shorten >=1.5pt},
  bc/.style={draw=black!70, fill=white, rounded corners=1pt, inner sep=3pt, align=center},
  wall/.style={black!82, line width=24pt, line cap=butt, line join=round},
  fluid/.style={blue!6, line width=20pt, line cap=butt, line join=round}
]

\def\r{1.65}
\def\w{0.38}
\def\xc{3.45}
\def\cy{1.75}
\def\xl{\xc-\r}
\def\xr{\xc+\r}
\def\ytop{6.00}

\coordinate (leftTop) at (\xl,\ytop);
\coordinate (leftBend) at (\xl,\cy);
\coordinate (rightBend) at (\xr,\cy);
\coordinate (rightTop) at (\xr,\ytop);
\coordinate (bendCenter) at (\xc,\cy);

% Vertical U-bend centerline: inlet top-left, outlet top-right.
\draw[wall] (leftTop) -- (leftBend)
  arc[start angle=180,end angle=360,radius=\r] -- (rightTop);
\draw[fluid] (leftTop) -- (leftBend)
  arc[start angle=180,end angle=360,radius=\r] -- (rightTop);

% Inlet and outlet planes.
\draw[red!70!black, line width=1.15pt] (\xl-\w,\ytop) -- (\xl+\w,\ytop);
\draw[red!70!black, line width=1.15pt] (\xr-\w,\ytop) -- (\xr+\w,\ytop);

% Flow direction arrows on centerline, away from walls.
\draw[flow] (\xl,5.45) -- (\xl,4.55);
\draw[flow] (\xl,3.85) -- (\xl,2.95);
\draw[flow] ({\xc+\r*cos(210)},{\cy+\r*sin(210)})
  arc[start angle=210,end angle=330,radius=\r];
\draw[flow] (\xr,2.95) -- (\xr,3.85);
\draw[flow] (\xr,4.55) -- (\xr,5.45);

% Radius of curvature.
\draw[dim] (bendCenter) -- node[right, fill=white, inner sep=1pt]
  {$R_c=0.125$ m} (\xc,{\cy-\r});
\fill[black] (bendCenter) circle (1.15pt);

% Pipe diameter, shown near the outlet leg to avoid the straight-leg dimension.
\draw[dim] (\xr-\w,5.55) -- node[below, fill=white, inner sep=1pt]
  {$D=0.028$ m} (\xr+\w,5.55);

% Extraction line at the start of the outlet straight section.
\draw[black!65, dashed, line width=0.8pt] (\xr-\w,\cy) -- (\xr+\w,\cy);
\node[bc, anchor=west, text width=33mm] (profileLabel) at (\xr+0.95,\cy)
  {bend-exit\\profile line $\Gamma_D$};
\draw[-{Latex[length=2.3mm,width=1.5mm]}, black!55, line width=0.65pt]
  (profileLabel.west) -- (\xr+\w,\cy);

% Boundary-condition labels.
\node[bc, anchor=south east, text width=32mm] at (\xl-0.35,\ytop+0.40)
  {inlet\\$\mathbf{u}_{in}=1.42$ m/s\\$\tilde{\nu}_{in}=9.34{\times}10^{-5}$};
\node[bc, anchor=south west, text width=25mm] at (\xr+0.35,\ytop+0.40)
  {outlet\\$p=0$ Pa};
\node[bc, anchor=west, text width=37mm] (wallLabel) at (\xr+0.95,3.65)
  {no-slip walls\\wall-resolved SA\\$Re_D\approx44{,}700$\\$De\approx15{,}000$};
\draw[-{Latex[length=2.3mm,width=1.5mm]}, black!55, line width=0.65pt]
  (wallLabel.west) -- (\xr,3.95);

\draw[dim] (\xl-0.92,\ytop) -- node[midway, rotate=90, above, fill=white, inner sep=1pt]
  {shortened straight leg, $L=30D$} (\xl-0.92,\cy);

\node[anchor=west, text=black!65, font=\sffamily\footnotesize] at (0.15,-0.42)
  {Flow enters at the top left, turns through the lower bend, and exits at the top right.};

\end{tikzpicture}
